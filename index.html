class ArticleController
{
    public function summarize($id)
    {
        $article = Article::find($id);

        $summary = ai()->ask(
            "Summarize this article in 2 sentences: $article->body"
        );

        return response()->json([
            'summary' => $summary,
        ]);
    }
}
